

---@generic T
---@param arr T[]
---@return T[]
local function copy(arr) end

---@generic T
---@generic U
---@param arr T[]
---@param func fun(value: T, key: int, arr: T[]): U
---@return U[]
local function map(arr, func) end

---@generic T
---@param arr T[]
---@param func fun(value: T, key: int, arr: T[]): bool
---@return T[]
local function filter(arr, func) end

---`arr`をコピーし、sortしたものを返却する。
---@generic T
---@param arr T[]
---@param comp? fun(a: T, b: T): bool `a`が`b`より「小さい」か否かを返却
---@return T[]
local function sorted(arr, comp) end

---`arr`を`n`個にコピーして結合したものを返却する。
---@generic T
---@param arr T[]
---@param n int
---@return T[]
local function rep(arr, n) end

---`st`から`st+n-1`までの`n`個の順列を生成する。
---@param n? int
---@param st? int default: `1`
---@return int[]
local function _seq(n, st) end
seq = _seq

---`st`から`ed`までの順列を生成する。
---@param st int
---@param ed int
---@return int[]
local function _range(st, ed) end
range = _range

---配列`arr`から要素`indices`を抽出する。
---@generic T
---@param arr T[]
---@param indices (int[] | int)[]
---@return T[]
local function _pick(arr, indices) end
pick = _pick

---@class CountUpObj
---@overload fun(value: int | string | int[] | string[]): CountUpObj
local CountUpObj = {}

---@return CountUpObj
function CountUpObj:reset() self.tbl = {} return self end
---@param value int | string | int[] | string[]
---@return CountUpObj
function CountUpObj:add(value) end
---@param value int | int[]
---@param digit? int
---@return CountUpObj
function CountUpObj:addHex(value, digit) end
---@param value string
---@return int
function CountUpObj:get(value) end
function CountUpObj:print() end

---@param arr? int[] | string[]
---@return CountUpObj
local function countUp(arr) end
CountUp = countUp

---配列の配列をすべてconcatする
---@generic T
---@param arrs T[][]
---@return T[]
local function flat(arrs) end

---2つの配列を結合する
---@generic T
---@param arr1 T[]
---@param arr2 T[]
---@return T[]
local function concat(arr1, arr2) end

---(破壊的変更)`arr`の`pos`位置に`added`配列を挿入する
---@generic T
---@param arr int[]
---@param added int[]
---@param pos? int
local function insert_array(arr, added, pos) end

---配列の`[st, ed)`部分を抽出する。 (添字は0始まり)
---@generic T
---@param arr T[]
---@param st int
---@param ed? int non-inclusive
---@return T[]
local function slice(arr, st, ed) end

Array = {
    copy = copy,
    map = map,
    filter = filter,
    sorted = sorted,
    rep = rep,
    flat = flat,
    concat = concat,
    seq = _seq,
    range = _range,
    pick = _pick,
    countUp = countUp,
    insert_array = insert_array,
    slice = slice,
}
__map = map
__filter = filter
__sorted = sorted
