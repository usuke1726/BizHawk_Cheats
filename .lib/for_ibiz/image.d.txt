
---@class ImageFilterObj
---@field _vs int[]
---@field _width int?
---@field _height int?
---@field _addr_bound int?
---@field protected _ready bool
local ImageFilterObj = {}

---@param width int
---@return ImageFilterObj
function ImageFilterObj:width(width) end
---@param bound int
---@return ImageFilterObj
function ImageFilterObj:bound(bound) end
---@param height int
---@return ImageFilterObj
function ImageFilterObj:height(height) end

---@param filter_func fun(x: int, y: int): int?, int?
---@return ImageFilterObj
function ImageFilterObj:go(filter_func) end

---VRAM上などのピクセルデータにフィルタリングをかけるオブジェクトを生成する。
---
---使用例：
--- ```lua
--- local W = 0x80
--- local function pos(x, y)
---     return lib.pos.Morton(x, y, 0x04, 0x20, 0x400) + 0x06214000
--- end
--- lib.image.filterObj()
--- :width(W):bound(0x6000)
--- :go(function(x, y)
---     local _x = math.round(x * 0.3) + (W // 2)
---     return pos(x, y), r8(pos(_x, y))
--- end)
--- ```
---@return ImageFilterObj
local function filterObj() end

---@class ImageLib
local __image_lib = {
    filterObj = filterObj,
}
