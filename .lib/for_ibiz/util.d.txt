---`min`を0、`max`を1として、`v`がどの位置に属するか(区間`[0,1]`)を計算する。
---
---ただし、`v`が`[min, max]`の外にいる場合は、`[0, 1]`の外の値が返却される。
---@param v number
---@param min number
---@param max number
---@return number
local function _invlerp(v, min, max)
    assert(max > min, "max must be greater than min")
    return (v - min) / (max - min)
end
lerp = _lerp
ilerp = _ilerp
invlerp = _invlerp
math.lerp = _lerp
math.ilerp = _ilerp
math.invlerp = _invlerp

---2次元座標をモートン順序に基づき1次元配列の座標に変換する。
---
--- - `W`: ブロック幅("W"idth, ブロック内の次の行に移行するまでの距離)
--- - `N`: 次のブロックに移行するまでの距離(つまりブロックのピクセル総数)
---     - ("N"eighbor, "N"ext block, "N"umber of pixels per block)
--- - `L`: 次のブロック行("L"ine)に移行するまでの距離
--- 
--- #### 注意
--- 
--- - `W | N | L`、つまり`N`は`W`の倍数、`L`は`N`の倍数でなければならない。
---
--- #### 図解 (`W = 4, N = 8, L = 16`)
--- 
--- ```plaintext
---             N
---      |-- -- -- ---|
---      |<-- W -->|  v  
---   -- 00 01 02 03 08 09 10 11
--- L |  04 05 06 07 12 13 14 15
---   -> 16 17 18 19 24 25 26 27
---      20 21 22 23 28 29 30 31
--- ```
---@param x int
---@param y int
---@param W int
---@param N int
---@param L int
---@return int
local function pos_Morton(x, y, W, N, L) end
