
---@enum DS_Keys
local button_keys = {
    A = "A",
    B = "B",
    X = "X",
    Y = "Y",
    L = "L",
    R = "R",
    Up = "Up",
    Down = "Down",
    Left = "Left",

---@param key DS_Keys
---@return bool
local function ds_key_get(key) end

---指定したキーの情報**のみ**の設定を上書きする。(他のキーの入力状態は変更しない)

---@overload fun(info: table<DS_Keys, bool | int>)
local function ds_key_set(key, on) end

local __ds_key_lib = {
    key = button_keys,
    get = ds_key_get,
    set = ds_key_set,
}
