
---@enum DS_Keys
local button_keys = {
    A = "A",
    B = "B",
    X = "X",
    Y = "Y",
    L = "L",
    R = "R",
    Up = "Up",
    Down = "Down",
    Left = "Left",
    Right = "Right",
    Start = "Start",
    Select = "Select",
}
DS_KEYS = button_keys

---@param key DS_Keys
---@return bool
local function ds_key_get(key) end

---指定したキーの情報**のみ**の設定を上書きする。(他のキーの入力状態は変更しない)
---@param key DS_Keys
---@param on bool | int `int`値は非ゼロが`true`扱い
---@overload fun(info: table<DS_Keys, bool | int>)
local function ds_key_set(key, on) end

---現在のマウスの位置を返却する。
---
--- - `x`の範囲: `[0, 255]`
--- - `y`の範囲: `[0, 191]` (下画面), `[-192, -1]` (上画面)
---@return {x: int, y: int}
local function ds_get_mouse_pos() end
---画面をタッチしているなら現在のマウスの位置を返却。そうでないなら`nil`を返却する。
---
--- - `x`の範囲: `[0, 255]`
--- - `y`の範囲: `[0, 191]` (下画面), `[-192, -1]` (上画面)
---@return {x: int, y: int}?
local function ds_mouse() end
---指定した座標をタッチする。(引数無しならタッチペンを離す)
---
---※この方法でタッチするには、`Config/Controllers` から、`Analog Controls` の `Touch X`, `Touch Y`を**unbind**しておく必要がある！
---
--- - `x`の範囲: `[0, 255]`
--- - `y`の範囲: `[0, 191]` (下画面), `[-192, -1]` (上画面)
---@param x int
---@param y int
---@overload fun()
local function ds_touch(x, y) end


---@class ButtonLisObj
---@field key DS_Keys
---@overload fun(): bool | nil
local ButtonLisObj = {}

--- キーを離した瞬間か否か
--- 
--- ※`update`をしてから呼び出すこと。
---@return bool
function ButtonLisObj:keyup() end

--- キーを押しているか否か(押した瞬間は含まれない)
--- 
--- ※`update`をしてから呼び出すこと。
---@return bool
function ButtonLisObj:holding() end

--- キーを離しているか否か(押した瞬間は含まれない)
--- 
--- ※`update`をしてから呼び出すこと。
---@return bool
function ButtonLisObj:released() end

--- キーを押した瞬間か否か
--- 
--- ※`update`をしてから呼び出すこと。
---@return bool
function ButtonLisObj:keydown() end

---`update`した際に自動で実行する処理を登録する
---@param event "up" | "down"
---@param callback fun()
---@return ButtonLisObj
function ButtonLisObj:on(event, callback) end

--- `holding`, `released` 状態に移行するまでの遅延フレーム数を設定する。
---@param delay int
---@return ButtonLisObj
function ButtonLisObj:delay(delay) end

---状態を更新する。(`__call`メタメソッドと同等)
---@return boolean|nil `nil`: 変化なし, `true/false`: `down/up`
function ButtonLisObj:update() end

--- ボタン入力の状態を管理するオブジェクトを生成する
--- 
--- 使用例1：
--- ```lua
--- local L = ButtonLis("L"):delay(60)
--- function loop()
---     if L() then
---         print("pressed now!")
---     end
---     if L:holding() then
---         print("holding")
---     end
--- end
--- ```
--- 
--- ---
--- 使用例2：
--- 
--- ```lua
--- local L = ButtonLis("L")
--- :on("down", function()
---     -- do something
--- end)
--- function loop() L() end
--- ```
---@param key DS_Keys
---@return ButtonLisObj
local function _ButtonLis(key) end
ButtonLis = _ButtonLis

local __ds_key_lib = {
    key = button_keys,
    get = ds_key_get,
    set = ds_key_set,
    lis = _ButtonLis,
    get_mouse_pos = ds_get_mouse_pos,
    mouse = ds_mouse,
    touch = ds_touch,
}
